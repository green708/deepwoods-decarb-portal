// ── Map API client record → local engagement row ─────────────────────────────

function apiClientToEngagement(
  c: ApiClient,
  statusOverrides: Record<string, ClientStatus>,
  screened: Set<string>,
): Client {
  const id = String(c.id);
  const base: ClientStatus = !c.is_active
    ? "Onboarding"
    : c.setup_complete !== "true"
    ? "Onboarding"
    : "Boundary Set";
  const status: ClientStatus = statusOverrides[id] ?? base;
  return {
    id,
    company: c.name,
    entity: c.slug.replace(/-/g, "").toUpperCase().slice(0, 3),
    industry: c.industry ?? "—",
    country: "—",
    onboardedDate: c.activated_at?.slice(0, 10) ?? c.created_at.slice(0, 10),
    fy: c.reporting_period ?? "FY",
    contactName: "—",
    contactEmail: "—",
    status,
    dataProgress:
      status === "Complete" || status === "Report Ready" ? 100
      : status === "Review" ? 90
      : status === "Data Collection" ? 50
      : c.setup_complete === "true" ? 10
      : 0,
    facilitiesCount: c.entity_count || 1,
    scope3Screened: screened.has(id),
  };
}

// ── Create Client modal ───────────────────────────────────────────────────────

function CreateClientModal({ open, onClose, onCreated }: {
  open: boolean;
  onClose: () => void;
  onCreated: () => void;
}) {
  const [form, setForm] = useState({ name: "", slug: "", currency: "INR", reporting_period: "FY" });
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (form.name) {
      const auto = form.name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "");
      setForm(f => ({ ...f, slug: auto }));
    }
  }, [form.name]);

  const handleSubmit = async () => {
    if (!form.name.trim() || !form.slug) return;
    setSubmitting(true);
    setError(null);
    try {
      await authFetch("clients.php", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          name: form.name.trim(),
          slug: form.slug,
          currency: form.currency,
          reporting_period: form.reporting_period,
        }),
      });
      toast.success(`${form.name} created — pending activation`);
      onCreated();
      onClose();
      setForm({ name: "", slug: "", currency: "INR", reporting_period: "FY" });
    } catch (e: unknown) {
      setError(e instanceof Error ? e.message : "Failed to create client");
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 px-4">
      <div className="w-full max-w-md rounded-2xl bg-white p-6 shadow-2xl">

            <X className="h-4 w-4" />

          <label className="mb-1.5 block text-xs font-semibold uppercase tracking-wider text-[var(--navy-2)]">Company name</label>

            value={form.name}

type PaymentStatus = "pending" | "paid" | "waived";

function InvoiceModal({ client, open, onClose }: {
  client: ActivationClient | null;
  open: boolean;
  onClose: () => void;
}) {
  const updateInvoice = useUpdateClientInvoice();
  const [form, setForm] = useState<{
    invoice_ref: string;
    payment_status: PaymentStatus;
    payment_date: string;
    payment_notes: string;
  }>({ invoice_ref: "", payment_status: "pending", payment_date: "", payment_notes: "" });

  useEffect(() => {
    if (client) {
      setForm({
        invoice_ref:    client.invoice_ref    ?? "",
        payment_status: client.payment_status ?? "pending",
        payment_date:   client.payment_date   ?? "",
        payment_notes:  client.payment_notes  ?? "",
      });
    }
  }, [client]);

  const handleSave = async () => {
    if (!client) return;
    try {
      const result = await updateInvoice.mutateAsync({
        client_id:      client.id,
        invoice_ref:    form.invoice_ref    || null,
        payment_status: form.payment_status,
        payment_date:   form.payment_date   || null,
        payment_notes:  form.payment_notes  || null,
      });
      if (result?.was_activation) {
        if (result.email_sent) {
          toast.success("Payment approved. Access activation email sent.");
        } else {
          toast.warning("Payment approved, but the access activation email could not be delivered.");
        }
      } else {
        toast.success("Invoice details saved");
      }
      onClose();
    } catch (e: unknown) {
      toast.error(e instanceof Error ? e.message : "Save failed");
    }
  };

  if (!open || !client) return null;
  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 px-4">
      <div className="w-full max-w-md rounded-2xl bg-white p-6 shadow-2xl">
        <div className="mb-5 flex items-center justify-between">
          <div>
            <h2 className="text-base font-bold text-[var(--navy)]">Invoice &amp; payment</h2>
            <p className="mt-0.5 text-xs text-slate-400">{client.name}</p>
          </div>
          <button onClick={onClose} className="rounded-lg p-1.5 text-slate-400 hover:bg-slate-100">
            <X className="h-4 w-4" />
          </button>
        </div>

        <div className="mb-4">
          <label className="mb-1.5 block text-xs font-semibold uppercase tracking-wider text-[var(--navy-2)]">Invoice reference</label>
          <input
            value={form.invoice_ref}
            onChange={e => setForm(f => ({ ...f, invoice_ref: e.target.value }))}
            placeholder="INV-2026-001"
            className="w-full rounded-lg border border-[var(--border)] px-3.5 py-2.5 text-sm text-[var(--navy)] outline-none focus:border-[#1E6F5C] focus:ring-2 focus:ring-[#1E6F5C]/15"

            {c.invoice_ref ? ` · ${c.invoice_ref}` : ""}
            {c.payment_date ? ` · ${new Date(c.payment_date).toLocaleDateString("en-IN")}` : ""}
          </p>
        </div>

        {/* Status pill */}
        <PayStatusPill status={c.payment_status ?? "pending"} />

        {/* Portal access badge */}
        {c.is_active ? (
          <span className="flex items-center gap-1 rounded-full bg-[#E6F4EF] px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-[#1E6F5C]">
            <Unlock className="h-3 w-3" /> Portal active
          </span>
        ) : (
          <span className="flex items-center gap-1 rounded-full bg-slate-100 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-slate-500">
            <Lock className="h-3 w-3" /> Locked
          </span>
        )}

        {/* Actions */}
        <div className="flex items-center gap-2">
          {!isPaid && (
            <>
              <button
                onClick={() => quickMark(c, "paid")}
                disabled={updateInvoice.isPending}
                className="flex items-center gap-1.5 rounded-lg bg-[#1E6F5C] px-3 py-1.5 text-xs font-semibold text-white hover:brightness-110 disabled:opacity-50"
              >
                <CheckCircle2 className="h-3 w-3" /> Mark Paid
              </button>
              <button
                onClick={() => quickMark(c, "waived")}
                disabled={updateInvoice.isPending}
                className="flex items-center gap-1.5 rounded-lg border border-[#93C5FD] bg-[#EFF6FF] px-3 py-1.5 text-xs font-semibold text-[#1D4ED8] hover:bg-[#DBEAFE] disabled:opacity-50"
              >
                Waive
              </button>
            </>
          )}
          {isPaid && (
            <button
              onClick={() => quickMark(c, "pending")}
              disabled={updateInvoice.isPending}
              className="flex items-center gap-1.5 rounded-lg border px-3 py-1.5 text-xs font-medium hover:bg-slate-50 disabled:opacity-50"
              style={{ borderColor: "var(--border)", color: "var(--navy-2)" }}
            >
              <XCircle className="h-3 w-3" /> Revert to Pending
            </button>
          )}
          <button
            onClick={() => setInvoiceTarget(c)}
            className="flex items-center gap-1.5 rounded-lg border px-3 py-1.5 text-xs font-medium hover:bg-slate-50"
            style={{ borderColor: "var(--border)", color: "var(--navy)" }}
          >
            <FileText className="h-3 w-3" /> Details
          </button>
        </div>
      </div>
    );
  }

  return (
    <>
      <InvoiceModal client={invoiceTarget} open={invoiceTarget !== null} onClose={() => setInvoiceTarget(null)} />

      <div className="space-y-5">
        {/* Summary strip */}
        <div className="grid grid-cols-3 gap-3">
          {[
            { label: "Total clients",     value: clients.length,    color: "var(--navy)" },
            { label: "Payment pending",   value: clients.filter(c => !c.payment_status || c.payment_status === "pending").length, color: "#D97706" },
            { label: "Payment complete",  value: clients.filter(c => c.payment_status === "paid" || c.payment_status === "waived").length, color: "#1E6F5C" },
          ].map(s => (
            <div key={s.label} className="rounded-xl border bg-white px-4 py-3 shadow-sm" style={{ borderColor: "var(--border)" }}>
              <p className="text-xs font-medium" style={{ color: "var(--navy-2)" }}>{s.label}</p>
              <p className="text-2xl font-bold mt-0.5" style={{ color: s.color }}>{s.value}</p>
            </div>
          ))}
        </div>

        {/* Search */}
        <div className="relative max-w-sm">
          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
          <input value={search} onChange={e => setSearch(e.target.value)}
            placeholder="Search clients…"
            className="w-full pl-9 pr-3 py-2 rounded-lg border text-sm"
            style={{ borderColor: "var(--border)" }} />
        </div>

        {isLoading && (
          <div className="flex items-center gap-2 text-sm text-muted-foreground">
            <Loader2 className="h-4 w-4 animate-spin" /> Loading clients…
          </div>
        )}

        {/* Pending section */}
        {pending.length > 0 && (
          <section>
            <div className="mb-2 flex items-center gap-2">
              <AlertCircle className="h-3.5 w-3.5 text-amber-600" />
              <span className="text-xs font-bold uppercase tracking-wider text-amber-600">
                Payment Pending ({pending.length})
              </span>
            </div>
            <div className="space-y-2">
              {pending.map(c => <ClientPayRow key={c.id} c={c} />)}
            </div>
          </section>
        )}

        {/* Completed section */}
        {completed.length > 0 && (
          <section>
            <div className="mb-2 flex items-center gap-2">
              <CheckCircle2 className="h-3.5 w-3.5 text-[#1E6F5C]" />
              <span className="text-xs font-bold uppercase tracking-wider text-[#1E6F5C]">
                Payment Complete ({completed.length})
              </span>
            </div>
            <div className="space-y-2">
              {completed.map(c => <ClientPayRow key={c.id} c={c} />)}
            </div>
          </section>
        )}

        {!isLoading && clients.length === 0 && (
          <div className="rounded-xl border border-dashed px-6 py-10 text-center text-sm text-slate-400"
            style={{ borderColor: "var(--border)" }}>
            No clients found.
          </div>
        )}
      </div>
    </>
  );
}

type ReqStatus = "new" | "contacted" | "converted" | "rejected";
interface InternalNote { text: string; added_at: string; }
interface ClientReq {
  id: number; name: string; company: string; email: string;
  phone: string; designation: string; employees: string; message: string; industry: string;
  is_india: number; gstin: string; legal_name: string; operating_name: string;
  org_type: string; is_msme: number | null; legal_address: string; website: string;
  ghg_done_before: boolean | null; ghg_requirement: string | null;
  services_needed: string[] | null; services_other: string | null;
  internal_notes: InternalNote[] | null;
  status: ReqStatus; created_at: string; updated_at: string;
}
interface ConvResult { slug: string; inviteEmail: string; tempPassword: string | null; emailSent: boolean; smtpDiag: string | null; }

const REQ_STATUS: Record<ReqStatus, { label: string; cls: string }> = {
  new:       { label: "New",       cls: "bg-amber-100 text-amber-700 border border-amber-200" },
  contacted: { label: "Contacted", cls: "bg-blue-100 text-blue-700 border border-blue-200" },
  converted: { label: "Converted", cls: "bg-[#E6F4EF] text-[#1E6F5C] border border-[#C5E4D0]" },
  rejected:  { label: "Rejected",  cls: "bg-slate-100 text-slate-500 border border-slate-200" },
};

function slugify(s: string) {
  return s.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
}

function RequestsTab() {
  const role = getRole();
  const isAdmin = role === "deepwoods_admin";
  const qc = useQueryClient();
  const [expanded, setExpanded] = useState<Set<number>>(new Set());
  const [convertTarget, setConvertTarget] = useState<ClientReq | null>(null);
  const [convertResult, setConvertResult] = useState<ConvResult | null>(null);
  const [emailResults, setEmailResults] = useState<Map<number, boolean>>(new Map());
  const [noteTexts, setNoteTexts] = useState<Map<number, string>>(new Map());
  const [emailTexts, setEmailTexts] = useState<Map<number, string>>(new Map());
  const [showEmailBox, setShowEmailBox] = useState<Set<number>>(new Set());
  const [sendingEmail, setSendingEmail] = useState<number | null>(null);
  const [emailSentFor, setEmailSentFor] = useState<Set<number>>(new Set());

  const { data: requests = [], isLoading, error } = useQuery<ClientReq[]>({
    queryKey: ["client-requests"],
    queryFn: () => authFetch("client-requests.php"),
  });

  const patchStatus = useMutation({
    mutationFn: ({ id, status }: { id: number; status: "contacted" | "rejected" }) =>
      authFetch("client-requests.php", {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ id, status }),
      }),
    onSuccess: () => qc.invalidateQueries({ queryKey: ["client-requests"] }),
    onError: (e: Error) => toast.error(e.message),
  });

  const addNote = useMutation({
    mutationFn: ({ id, note }: { id: number; note: string }) =>
      authFetch("client-requests.php", {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ id, note }),
      }),
    onSuccess: (_data, variables) => {
      setNoteTexts(prev => { const m = new Map(prev); m.set(variables.id, ""); return m; });
      qc.invalidateQueries({ queryKey: ["client-requests"] });
    },
    onError: (e: Error) => toast.error(e.message),
  });

  const toggleExpand = (id: number) =>
    setExpanded(prev => { const n = new Set(prev); n.has(id) ? n.delete(id) : n.add(id); return n; });

    setShowEmailBox(prev => { const n = new Set(prev); n.has(id) ? n.delete(id) : n.add(id); return n; });

        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ action: "send_email", id: req.id, message: msg }),
      });
      setEmailSentFor(prev => new Set(prev).add(req.id));
      setEmailTexts(prev => { const m = new Map(prev); m.set(req.id, ""); return m; });
      setShowEmailBox(prev => { const n = new Set(prev); n.delete(req.id); return n; });

      toast.error((e as Error).message);
    } finally {
      setSendingEmail(null);
    }
  };

    <div className="space-y-4">

            return (

                    ? <ChevronDown className="h-4 w-4 shrink-0 text-muted-foreground" />

                      <span className="text-xs text-muted-foreground">·</span>
                      <span className="text-xs text-muted-foreground">{req.name}</span>
                      {req.designation && (
                        <span className="text-xs text-muted-foreground">· {req.designation}</span>
                      )}
                    </div>
                    <div className="flex items-center gap-3 flex-wrap">
                      <span className="text-xs text-muted-foreground">{req.email}</span>
                      {notes.length > 0 && (
                        <span className="text-[10px] rounded-full px-2 py-0.5 bg-amber-50 border border-amber-200 text-amber-700 font-semibold">

                    {(services.length > 0 || req.ghg_done_before !== null) && (

                          >
                            {sendingEmail === req.id && <Loader2 className="h-3.5 w-3.5 animate-spin" />}

                        >
                          {addNote.isPending ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Plus className="h-3.5 w-3.5" />}
                          Add note
                        </button>
                      </div>
                    </div>

                    {req.status === "rejected" && (
                      <p className="text-xs text-muted-foreground">Rejection email sent. No account was created.</p>
                    )}
                    {req.status === "converted" && (
                      <p className="text-xs font-medium" style={{
                        color: !emailResults.has(req.id)
                          ? "#6B7280"
                          : emailResults.get(req.id) ? "#1E6F5C" : "#B45309",
                      }}>
                        {!emailResults.has(req.id)
                          ? "Converted to client — email delivery status unavailable."
                          : emailResults.get(req.id)
                            ? "Converted — invitation email sent successfully."
                            : "Converted — invitation email could not be sent."}
                      </p>
                    )}
                  </div>
                )}
              </div>
            );
          })}
        </div>
      )}

      {convertTarget && (
        <AcceptModal
          request={convertTarget}
          onClose={() => setConvertTarget(null)}
          onSuccess={(result) => {
            const reqId = convertTarget!.id;
            setConvertTarget(null);
            setConvertResult(result);
            setEmailResults(prev => new Map(prev).set(reqId, result.emailSent));
            qc.invalidateQueries({ queryKey: ["client-requests"] });
          }}
        />
      )}
      {convertResult && (
        <AcceptSuccessModal result={convertResult} onClose={() => setConvertResult(null)} />
      )}
    </div>
  );
}

function AcceptModal({ request, onClose, onSuccess }: {
  request: ClientReq;
  onClose: () => void;
  onSuccess: (r: ConvResult) => void;
}) {
  const [name, setName]   = useState(request.company);
  const [slug, setSlug]   = useState(slugify(request.company));
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const [slugTouched, setSlugTouched] = useState(false);
  const handleNameChange = (v: string) => {
    setName(v);
    if (!slugTouched) setSlug(slugify(v));
  };

  const inp = "w-full rounded-lg border border-[var(--border)] bg-white px-3 py-2 text-sm text-[var(--navy)] focus:border-[#1E6F5C] focus:outline-none";
  const lbl = "mb-1 block text-[10px] font-bold uppercase tracking-widest text-muted-foreground";

  const submit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);
    setLoading(true);
    try {
      const res = await authFetch("client-requests.php", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          action: "convert",
          id: request.id,
          name,
          slug,
          invite_email: request.email,
          invite_role: "sustainability_team",
        }),
      });
      onSuccess({
        slug: res.slug,
        inviteEmail: res.invite_email ?? request.email,
        tempPassword: res.temp_password ?? null,
        emailSent: !!res.email_sent,
        smtpDiag: res.smtp_diag ?? null,
      });
    } catch (err) {
      setError(err instanceof Error ? err.message : "Approval failed");
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 p-4">
      <div className="w-full max-w-md rounded-xl bg-card p-6 shadow-2xl">
        <div className="mb-4 flex items-start justify-between">
          <div>
            <h3 className="text-lg font-bold text-[var(--navy)]">Approve & Invite</h3>
            <p className="mt-0.5 text-xs text-muted-foreground">
              Creates the account and sends login credentials to <strong>{request.email}</strong> in one step.
            </p>
          </div>
          <button type="button" onClick={onClose} className="rounded p-1 hover:bg-slate-100">
            <X className="h-4 w-4" />
          </button>
        </div>

        {/* What happens summary */}
        <div className="mb-4 rounded-lg border border-[#C5E4D0] bg-[#F0FAF5] px-3 py-2.5 text-xs text-[#1E6F5C] space-y-1">
          <p className="font-semibold">This will automatically:</p>
          <p>1. Create the client account</p>
          <p>2. Create login for <strong>{request.email}</strong> (Sustainability Team role)</p>
          <p>3. Generate a temporary password and send credentials by email</p>
          <p>4. Mark this request as accepted</p>
        </div>
        <div className="mb-4 rounded-lg border border-amber-200 bg-amber-50 px-3 py-2 text-xs text-amber-700">
          After approval, go to <strong>Entity Codes</strong> tab to set a unique entity code for this client.
        </div>

        {error && (
          <div className="mb-4 rounded-lg border border-red-200 bg-red-50 px-3 py-2 text-xs text-red-700">{error}</div>
        )}

        <form onSubmit={submit} className="space-y-3">
          <div>
            <label className={lbl}>Client name *</label>
            <input value={name} onChange={e => handleNameChange(e.target.value)} required className={inp} />
          </div>
          <div>
            <label className={lbl}>Slug * <span className="normal-case font-normal text-muted-foreground">(URL identifier)</span></label>
            <input
              value={slug}
              onChange={e => { setSlugTouched(true); setSlug(e.target.value); }}
              required pattern="[-a-z0-9]+"
              className={inp}
            />
            <p className="mt-0.5 text-[10px] text-muted-foreground">Lowercase letters, numbers and hyphens only</p>
          </div>
          <div className="flex justify-end gap-2 pt-2">
            <button type="button" onClick={onClose} className="rounded-lg border px-4 py-2 text-sm hover:bg-slate-50">Cancel</button>
            <button type="submit" disabled={loading || !name.trim() || !slug}
              className="flex items-center gap-2 rounded-lg bg-[#1E6F5C] px-5 py-2 text-sm font-semibold text-white hover:brightness-110 disabled:opacity-50">
              {loading && <Loader2 className="h-4 w-4 animate-spin" />}
              {loading ? "Approving…" : "Approve & Invite"}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

// ── Verification tab component (ISO 14064-3 / CDP C10 / GRI 2-5) ─────────────

type AssuranceLevel = "limited" | "reasonable" | "none";
type AssuranceStandard = "ISAE3410" | "ISO14064-3" | "AA1000AS" | "other";

interface VerificationRecord {
  reportingYear: string;
  assuranceLevel: AssuranceLevel;
  standard: AssuranceStandard;
  provider: string;
  scopeCovered: string;
  savedAt?: string;
}

function VerificationTab() {
  const [records, setRecords] = useState<VerificationRecord[]>([]);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [form, setForm] = useState<VerificationRecord>({
    reportingYear: "",
    assuranceLevel: "limited",
    standard: "ISAE3410",
    provider: "",
    scopeCovered: "",
  });

  useEffect(() => {
    authFetch("verification-status.php")
      .then((d: { records: VerificationRecord[] }) => setRecords(d.records ?? []))
      .catch(() => {})
      .finally(() => setLoading(false));
  }, []);

  const handleSave = async () => {
    if (!form.reportingYear || !form.provider || !form.scopeCovered) {
      toast.error("Reporting year, provider, and scope covered are required.");
      return;
    }
    setSaving(true);
    try {
      const updated = [
        ...records.filter(r => r.reportingYear !== form.reportingYear),
        { ...form, savedAt: new Date().toISOString().slice(0, 10) },
      ].sort((a, b) => b.reportingYear.localeCompare(a.reportingYear));
      await authFetch("verification-status.php", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ records: updated }),
      });
      setRecords(updated);
      toast.success("Verification record saved.");
      setForm({ reportingYear: "", assuranceLevel: "limited", standard: "ISAE3410", provider: "", scopeCovered: "" });
    } catch (e: unknown) {
      toast.error(e instanceof Error ? e.message : "Save failed");
    } finally {
      setSaving(false);
    }
  };

  const LEVEL_LABELS: Record<AssuranceLevel, string> = {
    limited: "Limited Assurance",
    reasonable: "Reasonable Assurance",
    none: "No Assurance",
  };
  const STD_LABELS: Record<AssuranceStandard, string> = {
    ISAE3410: "ISAE 3410",
    "ISO14064-3": "ISO 14064-3",
    AA1000AS: "AA1000AS",
    other: "Other",
  };

  return (
    <div className="space-y-6">
      <div>
        <h3 className="text-base font-bold mb-1" style={{ color: "var(--navy)" }}>Third-Party Verification / Assurance</h3>
        <p className="text-sm" style={{ color: "var(--navy-2)" }}>
          Required by CDP C10, ISO 14064-3, and GRI 2-5. Record the assurance provider, level, and standard for each reporting year.
          An "Assured" badge will appear on the client dashboard once a record exists.
        </p>
      </div>

      {/* Add / edit record */}
      <div className="bg-white rounded-xl border p-5 shadow-sm" style={{ borderColor: "var(--border)" }}>
        <p className="text-xs font-bold uppercase tracking-wider mb-4" style={{ color: "var(--navy-2)" }}>Add / Update Record</p>
        <div className="grid sm:grid-cols-2 gap-4">
          <div>
            <label className="block text-xs font-semibold mb-1.5" style={{ color: "var(--navy-2)" }}>Reporting Year *</label>
            <input
              className="w-full rounded-lg border px-3.5 py-2 text-sm"
              style={{ borderColor: "var(--border)" }}
              placeholder="e.g. FY25"
              value={form.reportingYear}
              onChange={e => setForm(f => ({ ...f, reportingYear: e.target.value }))}
            />
          </div>
          <div>
            <label className="block text-xs font-semibold mb-1.5" style={{ color: "var(--navy-2)" }}>Assurance Level *</label>
            <select
              className="w-full rounded-lg border px-3.5 py-2 text-sm"
              style={{ borderColor: "var(--border)" }}
              value={form.assuranceLevel}
              onChange={e => setForm(f => ({ ...f, assuranceLevel: e.target.value as AssuranceLevel }))}
            >
              <option value="limited">Limited Assurance</option>
              <option value="reasonable">Reasonable Assurance</option>
              <option value="none">No Assurance (self-declaration)</option>
            </select>
          </div>
          <div>
            <label className="block text-xs font-semibold mb-1.5" style={{ color: "var(--navy-2)" }}>Assurance Standard *</label>
            <select
              className="w-full rounded-lg border px-3.5 py-2 text-sm"
              style={{ borderColor: "var(--border)" }}
              value={form.standard}
              onChange={e => setForm(f => ({ ...f, standard: e.target.value as AssuranceStandard }))}
            >
              <option value="ISAE3410">ISAE 3410</option>
              <option value="ISO14064-3">ISO 14064-3</option>
              <option value="AA1000AS">AA1000AS</option>
              <option value="other">Other</option>
            </select>
          </div>
          <div>
            <label className="block text-xs font-semibold mb-1.5" style={{ color: "var(--navy-2)" }}>Provider Name *</label>
            <input
              className="w-full rounded-lg border px-3.5 py-2 text-sm"
              style={{ borderColor: "var(--border)" }}
              placeholder="e.g. Ernst & Young LLP"
              value={form.provider}
              onChange={e => setForm(f => ({ ...f, provider: e.target.value }))}
            />
          </div>
          <div className="sm:col-span-2">
            <label className="block text-xs font-semibold mb-1.5" style={{ color: "var(--navy-2)" }}>Scope Covered *</label>
            <input
              className="w-full rounded-lg border px-3.5 py-2 text-sm"
              style={{ borderColor: "var(--border)" }}
              placeholder="e.g. Scope 1 and 2 for all consolidated entities; Scope 3 Cat 1, 3, 6"
              value={form.scopeCovered}
              onChange={e => setForm(f => ({ ...f, scopeCovered: e.target.value }))}
            />
          </div>
        </div>
        <button
          onClick={handleSave}
          disabled={saving}
          className="mt-4 flex items-center gap-2 rounded-lg bg-[#1E6F5C] px-4 py-2 text-sm font-semibold text-white hover:brightness-110 disabled:opacity-50"
        >
          {saving ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <ShieldCheck className="w-3.5 h-3.5" />}
          Save Record
        </button>
      </div>

      {/* Existing records */}
      {loading ? (
        <p className="text-sm" style={{ color: "var(--navy-2)" }}>Loading…</p>
      ) : records.length === 0 ? (
        <p className="text-sm" style={{ color: "var(--navy-2)" }}>No verification records yet.</p>
      ) : (
        <div className="space-y-2">
          {records.map(r => (
            <div key={r.reportingYear} className="bg-white rounded-xl border px-5 py-4 flex flex-wrap gap-4 items-center shadow-sm" style={{ borderColor: "var(--border)" }}>
              <div className="flex items-center gap-2 shrink-0">
                <ShieldCheck className="w-4 h-4" style={{ color: "#1E6F5C" }} />
                <span className="text-sm font-bold" style={{ color: "var(--navy)" }}>{r.reportingYear}</span>
              </div>
              <span className="text-xs font-medium px-2.5 py-1 rounded-full bg-emerald-50 text-emerald-700 border border-emerald-200">{LEVEL_LABELS[r.assuranceLevel]}</span>
              <span className="text-xs font-mono" style={{ color: "var(--navy-2)" }}>{STD_LABELS[r.standard]}</span>
              <span className="text-sm" style={{ color: "var(--navy-2)" }}>{r.provider}</span>
              <span className="text-xs" style={{ color: "var(--slate)", flex: 1 }}>{r.scopeCovered}</span>
              {r.savedAt && <span className="text-xs shrink-0" style={{ color: "var(--slate)" }}>Saved {r.savedAt}</span>}
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

interface BrandingClient { id: number; name: string; logo_url: string | null; primary_color: string; }

function BrandingTab() {
  const qc = useQueryClient();

  const { data: raw = [] } = useQuery<ActivationClient[]>({
    queryKey: ["activation-clients"],
    queryFn: () => authFetch("activate-client.php"),
  });

  const [editing, setEditing] = useState<number | null>(null);
  const [logoUrl, setLogoUrl] = useState("");
  const [color, setColor]   = useState("#1E6F5C");
  const [saving, setSaving] = useState(false);
  const logoRef = useRef<HTMLInputElement>(null);

  const clients: BrandingClient[] = raw.map(c => ({
    id: c.id, name: c.name,
    logo_url: ((c as unknown) as BrandingClient).logo_url ?? null,
    primary_color: ((c as unknown) as BrandingClient).primary_color ?? "#1E6F5C",
  }));

  const openEdit = (c: BrandingClient) => {
    setEditing(c.id);
    setLogoUrl(c.logo_url ?? "");
    setColor(c.primary_color ?? "#1E6F5C");
  };

  const handleLogoFile = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = ev => setLogoUrl(ev.target?.result as string ?? "");
    reader.readAsDataURL(file);
  };

  const save = async () => {
    if (!editing) return;
    setSaving(true);
    try {
      await authFetch("branding.php", {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ client_id: editing, logo_url: logoUrl || null, primary_color: color }),
      });
      toast.success("Branding updated");
      qc.invalidateQueries({ queryKey: ["activation-clients"] });
      setEditing(null);
    } catch (e: unknown) {
      toast.error(e instanceof Error ? e.message : "Save failed");
    } finally {
      setSaving(false);
    }
  };

  const initials = (name: string) =>
    name.trim().split(/\s+/).slice(0, 2).map(w => w[0]).join("").toUpperCase();

  return (
    <div className="space-y-3">
      <p className="text-sm" style={{ color: "var(--navy-2)" }}>
        Set each client's logo and primary brand colour. These appear in their portal sidebar and reports.
      </p>

      {clients.map(c => (
        <div key={c.id} className="rounded-xl border bg-white p-4 shadow-sm" style={{ borderColor: "var(--border)" }}>
          <div className="flex items-center gap-4">
            {/* Logo preview */}
            <div className="flex-shrink-0 w-12 h-12 rounded-lg overflow-hidden border flex items-center justify-center bg-[#F9FAFB]"
              style={{ borderColor: "var(--border)" }}>
              {c.logo_url
                ? <img src={c.logo_url} alt={c.name} className="w-full h-full object-contain" />
                : <span className="text-sm font-bold" style={{ color: c.primary_color }}>{initials(c.name)}</span>
              }
            </div>

            {/* Client name + colour swatch */}
            <div className="flex-1 min-w-0">
              <p className="text-sm font-semibold truncate" style={{ color: "var(--navy)" }}>{c.name}</p>
              <div className="flex items-center gap-2 mt-1">
                <div className="w-4 h-4 rounded-full border" style={{ background: c.primary_color, borderColor: "var(--border)" }} />
                <span className="text-xs font-mono" style={{ color: "var(--navy-2)" }}>{c.primary_color}</span>
                {!c.logo_url && <span className="text-xs" style={{ color: "var(--navy-2)" }}>· No logo set</span>}
              </div>
            </div>

            <button onClick={() => editing === c.id ? setEditing(null) : openEdit(c)}
              className="flex-shrink-0 rounded-lg border px-3 py-1.5 text-xs font-semibold hover:bg-[#F3F4F6] transition-colors"
              style={{ borderColor: "var(--border)", color: "var(--navy)" }}>
              {editing === c.id ? "Cancel" : "Customise"}
            </button>
          </div>

          {/* Inline editor */}
          {editing === c.id && (
            <div className="mt-4 pt-4 border-t space-y-4" style={{ borderColor: "var(--border)" }}>
              <input ref={logoRef} type="file" accept="image/png,image/jpeg,image/svg+xml,image/webp"
                className="hidden" onChange={handleLogoFile} />

              {/* Logo */}
              <div>
                <label className="mb-1.5 block text-xs font-semibold uppercase tracking-wider" style={{ color: "var(--navy-2)" }}>
                  Logo
                </label>
                <div className="flex items-center gap-3">
                  <div className="w-12 h-12 rounded-lg border flex items-center justify-center bg-[#F9FAFB] flex-shrink-0"
                    style={{ borderColor: "var(--border)" }}>
                    {logoUrl
                      ? <img src={logoUrl} alt="" className="w-full h-full object-contain rounded-lg" />
                      : <span className="text-sm font-bold" style={{ color: color }}>{initials(c.name)}</span>
                    }
                  </div>
                  <div className="flex-1 min-w-0 space-y-2">
                    <input value={logoUrl} onChange={e => setLogoUrl(e.target.value)}
                      placeholder="https://company.com/logo.png"
                      className="w-full rounded-lg border px-3 py-2 text-sm outline-none focus:border-[#1E6F5C]"
                      style={{ borderColor: "var(--border)" }} />
                    <button type="button" onClick={() => logoRef.current?.click()}
                      className="text-xs font-semibold" style={{ color: "#1E6F5C" }}>
                      ↑ Or upload from computer
                    </button>
                  </div>
                  {logoUrl && (
                    <button type="button" onClick={() => setLogoUrl("")}
                      className="flex-shrink-0 rounded-lg border p-1.5 hover:bg-slate-100"
                      style={{ borderColor: "var(--border)", color: "var(--navy-2)" }}>

// ── Data Export modal ─────────────────────────────────────────────────────────

function DataExportModal({ client, open, onClose }: {
  client: ActivationClient | null;
  open: boolean;
  onClose: () => void;
}) {
  const [entity, setEntity] = useState("");
  const [fy, setFy]         = useState("FY26");
  const [busy, setBusy]     = useState(false);

  useEffect(() => { if (open) { setEntity(""); setFy("FY26"); } }, [open]);

  const download = async () => {
    if (!client || !entity.trim() || !fy.trim()) return;
    setBusy(true);
    try {
      // authFetch returns parsed JSON by default; for file downloads we need raw Response
      const token = getToken() ?? "";
      const response = await fetch(
        `/api/data-tracker/export.php?client_id=${client.id}&entity=${encodeURIComponent(entity.trim())}&fy=${encodeURIComponent(fy.trim())}`,
        { headers: { Authorization: `Bearer ${token}` } },
      );
      if (!response.ok) { toast.error("Export failed — check entity code and FY"); return; }
      const blob = await response.blob();
      const url  = URL.createObjectURL(blob);
      const a    = document.createElement("a");
      a.href     = url;
      a.download = `tracker_${entity.trim()}_${fy.trim()}_${new Date().toISOString().slice(0,10)}.csv`;
      a.click();
      URL.revokeObjectURL(url);
      onClose();
    } catch {
      toast.error("Download failed");
    } finally {
      setBusy(false);
    }
  };

  if (!open || !client) return null;
  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 px-4">
      <div className="w-full max-w-sm rounded-2xl bg-white p-6 shadow-2xl">
        <div className="mb-5 flex items-center justify-between">
          <div>
            <h2 className="text-base font-bold text-[var(--navy)]">Export tracker data</h2>
            <p className="mt-0.5 text-xs text-slate-400">{client.name}</p>
          </div>
          <button onClick={onClose} className="rounded-lg p-1.5 text-slate-400 hover:bg-slate-100"><X className="h-4 w-4" /></button>
        </div>
        <div className="mb-4">
          <label className="mb-1.5 block text-xs font-semibold uppercase tracking-wider text-[var(--navy-2)]">Entity code</label>
          <input
            value={entity}
            onChange={e => setEntity(e.target.value.toUpperCase())}
            placeholder="e.g. FP, RK, HQ"
            className="w-full rounded-lg border border-[var(--border)] px-3.5 py-2.5 text-sm text-[var(--navy)] outline-none focus:border-[#1E6F5C] focus:ring-2 focus:ring-[#1E6F5C]/15"
          />
        </div>
        <div className="mb-5">
          <label className="mb-1.5 block text-xs font-semibold uppercase tracking-wider text-[var(--navy-2)]">Fiscal year</label>
          <input
            value={fy}
            onChange={e => setFy(e.target.value)}
            placeholder="FY26"
            className="w-full rounded-lg border border-[var(--border)] px-3.5 py-2.5 text-sm text-[var(--navy)] outline-none focus:border-[#1E6F5C] focus:ring-2 focus:ring-[#1E6F5C]/15"
          />
        </div>
        <div className="flex gap-3">
          <button onClick={onClose} className="flex-1 rounded-lg border border-[var(--border)] py-2.5 text-sm font-medium text-[var(--navy)] hover:bg-slate-50">Cancel</button>
          <button
            onClick={download}
            disabled={!entity.trim() || !fy.trim() || busy}
            className="flex flex-1 items-center justify-center gap-2 rounded-lg bg-[#1E6F5C] py-2.5 text-sm font-semibold text-white disabled:opacity-40 hover:brightness-110"
          >
            {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <FileText className="h-4 w-4" />}
            Download CSV
          </button>
        </div>
      </div>
    </div>
  );
}

// ── Ledger Import tab ─────────────────────────────────────────────────────────

function LedgerImportTab({ clients }: { clients: ApiClient[] }) {
  const qc = useQueryClient();
  const fileRef  = useRef<HTMLInputElement>(null);
  const [clientId, setClientId]   = useState<number | null>(null);
  const [entity,   setEntity]     = useState("");
  const [preview,  setPreview]    = useState<LedgerParseResult | null>(null);
  const [file,     setFile]       = useState<File | null>(null);
  const [parsing,  setParsing]    = useState(false);
  const [importing,setImporting]  = useState(false);
  const [confirm,  setConfirm]    = useState(false);

  const selectedClient = clients.find(c => c.id === clientId) ?? null;

  const handleFile = useCallback(async (f: File) => {
    if (!clientId || !entity.trim()) {
      toast.error("Select a client and enter an entity code first");
      return;
    }
    setParsing(true);
    try {
      const parsed = await parseLedger(f, entity.trim().toUpperCase(), selectedClient?.reporting_period === "CY" ? "CY" : "FY", clientId);
      if (!parsed.year) { toast.error("Could not detect fiscal year — is this the correct template?"); return; }
      setFile(f);
      setPreview(parsed);
      setConfirm(true);
    } catch (e) {
      toast.error(`Parse error: ${e instanceof Error ? e.message : String(e)}`);
    } finally {
      setParsing(false);
      if (fileRef.current) fileRef.current.value = "";
    }
  }, [clientId, entity, selectedClient]);

  const doImport = async () => {
    if (!preview || !clientId) return;
    setImporting(true);
    try {
      const body = {
        target_client_id: clientId,
        entity: preview.entity,
        year:   preview.year,
        ghg_inventory:    preview.ghgInventory,
        s1_by_facility:   preview.s1ByFacility,
        s2_by_facility:   preview.s2ByFacility,
        s2_breakdown:     preview.s2Breakdown,
        scope3_summary:   preview.scope3Summary,
        intensity_metrics: preview.intensityMetrics,
        tracker_assets:   [],
        tracker_drafts:   [],
        irec_register:    preview.irecRegister ?? [],
        decarb_actions:   preview.decarbActions ?? [],
        peer_data:        preview.peerData ?? [],
      };
      const res = await authFetch("import/ledger.php", { method: "POST", body: JSON.stringify(body) });
      const c = res.counts ?? {};
      toast.success(`Imported ${preview.entity} ${preview.year} → ${c.ghg_inventory ?? 0} GHG rows, ${c.s1_by_facility ?? 0} S1 facilities, ${c.s2_by_facility ?? 0} S2 facilities`);
      setConfirm(false); setPreview(null); setFile(null);
      qc.invalidateQueries({ queryKey: ["ghg-inventory"] });
    } catch (e) {
      toast.error(`Import failed: ${e instanceof Error ? e.message : String(e)}`);
    } finally {
      setImporting(false);
    }
  };

  return (
    <div className="space-y-6">
      <div className="rounded-xl border border-[#C5E4D0] bg-[#F0FAF5] px-4 py-3.5">
        <p className="text-xs font-semibold text-[#1E6F5C]">Post-calculation import</p>
        <p className="mt-0.5 text-xs text-[#2D6A4F]">
          After completing GHG calculations offline, upload the filled client template here.
          This populates the client's dashboards, overview, inventory and all reports — it does <strong>not</strong> touch their submission or tracker data.
        </p>
      </div>

      {/* Client + entity selectors */}
      <div className="rounded-xl border bg-white p-5 space-y-4" style={{ borderColor: "var(--border)" }}>
        <h3 className="text-sm font-bold" style={{ color: "var(--navy)" }}>1. Select client &amp; entity</h3>
        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
          <div>
            <label className="mb-1.5 block text-xs font-semibold uppercase tracking-wider" style={{ color: "var(--navy-2)" }}>Client</label>
            <select
              value={clientId ?? ""}
              onChange={e => { setClientId(e.target.value ? Number(e.target.value) : null); setEntity(""); setPreview(null); }}
              className="w-full rounded-lg border px-3 py-2.5 text-sm outline-none focus:border-[#1E6F5C]"
              style={{ borderColor: "var(--border)" }}
            >
              <option value="">— select client —</option>
              {clients.map(c => <option key={c.id} value={c.id}>{c.name}</option>)}
            </select>
          </div>
          <div>
            <label className="mb-1.5 block text-xs font-semibold uppercase tracking-wider" style={{ color: "var(--navy-2)" }}>Entity code</label>
            <input
              value={entity}
              onChange={e => { setEntity(e.target.value.toUpperCase()); setPreview(null); }}
              placeholder="e.g. FP, HQ, ALL"
              disabled={!clientId}
              className="w-full rounded-lg border px-3 py-2.5 text-sm outline-none focus:border-[#1E6F5C] disabled:opacity-50"
              style={{ borderColor: "var(--border)" }}
            />
          </div>
        </div>
      </div>

      {/* File upload */}
      <div className="rounded-xl border bg-white p-5 space-y-3" style={{ borderColor: "var(--border)" }}>
        <h3 className="text-sm font-bold" style={{ color: "var(--navy)" }}>2. Upload calculated results template</h3>
        <label
          className={`flex cursor-pointer flex-col items-center gap-3 rounded-xl border-2 border-dashed px-6 py-10 transition ${!clientId || !entity.trim() ? "opacity-40 pointer-events-none" : "hover:border-[#1E6F5C] hover:bg-[#F0FAF5]"}`}
          style={{ borderColor: "var(--border)" }}
        >
          {parsing ? <Loader2 className="h-8 w-8 animate-spin text-[#1E6F5C]" /> : <FileUp className="h-8 w-8" style={{ color: "var(--navy-2)" }} />}

          </div>
          <input ref={fileRef} type="file" accept=".xlsx,.xls" className="hidden"
            disabled={!clientId || !entity.trim() || parsing}
            onChange={e => { const f = e.target.files?.[0]; if (f) handleFile(f); }} />
        </label>
      </div>

      {/* Confirm dialog */}
      {confirm && preview && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 px-4">
          <div className="w-full max-w-lg rounded-2xl bg-white p-6 shadow-2xl">

              <CheckCircle2 className="h-6 w-6 text-[#1E6F5C] shrink-0" />

                <h2 className="text-base font-bold text-[var(--navy)]">Confirm import</h2>
                <p className="text-xs text-slate-400">{selectedClient?.name} · {preview.entity} · {preview.year}</p>
              </div>
            </div>
            <div className="mb-5 grid grid-cols-2 gap-2 text-sm">
              {[
                ["GHG inventory rows",  preview.ghgInventory.length],
                ["S1 facility rows",    preview.s1ByFacility.length],
                ["S2 facility rows",    preview.s2ByFacility.length],
                ["Scope 3 categories",  preview.scope3Summary.length],
                ["I-REC rows",          preview.irecRegister?.length ?? 0],
                ["Decarb actions",      preview.decarbActions?.length ?? 0],
                ["Peers",               preview.peerData?.length ?? 0],
              ].map(([label, val]) => (

                  <span className={`text-sm font-bold ${Number(val) > 0 ? "text-[#1E6F5C]" : "text-slate-300"}`}>{String(val)}</span>
                </div>
              ))}
            </div>
            {preview.templateFormat === "client-template" && (
              <div className="mb-4 rounded-lg border border-[#C5E4D0] bg-[#F0FAF5] px-3 py-2 text-xs text-[#1E6F5C] font-medium">
                Client template format detected
              </div>
            )}
            <div className="flex gap-3">
              <button onClick={() => setConfirm(false)} disabled={importing}
                className="flex-1 rounded-lg border py-2.5 text-sm font-medium text-[var(--navy)] hover:bg-slate-50 disabled:opacity-40"
                style={{ borderColor: "var(--border)" }}>Cancel</button>
              <button onClick={doImport} disabled={importing}
                className="flex flex-1 items-center justify-center gap-2 rounded-lg bg-[#1E6F5C] py-2.5 text-sm font-semibold text-white disabled:opacity-40 hover:brightness-110">

        </div>
      )}
    </div>
  );
}

    setEditingId(e.id);
    setEditCode(e.code);
    setEditName(e.short_name || e.entity_name);
    setEditOrgType(e.org_type || '');
    setSaveError(null);
  };

  const cancelEdit = () => { setEditingId(null); setSaveError(null); };

  const saveEdit = async (id: number) => {
    setSaving(true);
    setSaveError(null);
    try {
      await authFetch('admin-entities.php', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ id, code: editCode, short_name: editName, org_type: editOrgType }),
      });
      qc.invalidateQueries({ queryKey: ['admin-entities'] });
      setEditingId(null);
      toast.success('Entity code updated');
    } catch (err) {
      setSaveError(err instanceof Error ? err.message : 'Save failed');
    } finally {
      setSaving(false);
    }
  };

  const inp = "rounded-lg border border-[var(--border)] bg-white px-2.5 py-1.5 text-sm text-[var(--navy)] focus:border-[#1E6F5C] focus:outline-none";

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between flex-wrap gap-3">
        <div>
          <h3 className="text-base font-semibold text-[var(--navy)]">Client Entities</h3>
          <p className="text-xs text-muted-foreground mt-0.5">
            Set the entity code used in facility & asset slugs. Each client should have a distinct code.
          </p>
        </div>
        <div className="relative max-w-xs w-full">
          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-muted-foreground" />
          <input value={search} onChange={e => setSearch(e.target.value)} placeholder="Search clients or codes…"
            className="w-full pl-8 pr-3 py-2 rounded-lg border border-[var(--border)] text-sm focus:outline-none focus:border-[#1E6F5C]" />
        </div>
      </div>

      {entities.some(e => e.code_duplicate) && (
        <div className="rounded-lg border border-amber-200 bg-amber-50 px-4 py-3 text-sm text-amber-800 flex gap-2 items-start">
          <AlertCircle className="h-4 w-4 mt-0.5 flex-shrink-0" />
          <span>Some entity codes are used by more than one client (highlighted in orange). Edit them to make each code unique.</span>
        </div>
      )}

      {isLoading ? (
        <div className="flex justify-center py-10"><Loader2 className="h-5 w-5 animate-spin text-muted-foreground" /></div>
      ) : (
        <div className="rounded-xl border border-[var(--border)] bg-card shadow-sm overflow-hidden">
          <table className="w-full text-sm">
            <thead className="border-b border-[var(--border)] bg-[#F9FAFB]">
              <tr>
                <th className="px-4 py-2.5 text-left text-[10px] font-bold uppercase tracking-widest text-muted-foreground">Client</th>
                <th className="px-4 py-2.5 text-left text-[10px] font-bold uppercase tracking-widest text-muted-foreground">Entity code</th>
                <th className="px-4 py-2.5 text-left text-[10px] font-bold uppercase tracking-widest text-muted-foreground">Entity name</th>
                <th className="px-4 py-2.5 text-left text-[10px] font-bold uppercase tracking-widest text-muted-foreground">Type</th>
                <th className="px-2 py-2.5" />
              </tr>
            </thead>
            <tbody>
              {filtered.map((ent, idx) => (
                <tr key={ent.id}
                  className={`${idx > 0 ? 'border-t border-[var(--border)]' : ''} ${ent.code_duplicate ? 'bg-amber-50/60' : ''}`}>
                  {editingId === ent.id ? (
                    <>
                      <td className="px-4 py-2.5 text-xs font-medium text-[var(--navy)]">{ent.client_name}</td>
                      <td className="px-4 py-2.5">
                        <input
                          value={editCode}
                          onChange={e => setEditCode(e.target.value.toUpperCase().replace(/[^A-Z0-9]/g, ''))}
                          maxLength={10}
                          className={inp + ' w-28 font-mono'}
                          autoFocus
                        />
                      </td>
                      <td className="px-4 py-2.5">
                        <input
                          value={editName}
                          onChange={e => setEditName(e.target.value)}
                          className={inp + ' w-48'}
                          placeholder="Short display name"
                        />
                      </td>
                      <td className="px-4 py-2.5">
                        <input
                          value={editOrgType}
                          onChange={e => setEditOrgType(e.target.value)}
                          className={inp + ' w-40'}
                          placeholder="e.g. Private Limited"
                        />
                      </td>
                      <td className="px-2 py-2.5">
                        <div className="flex items-center gap-1">
                          {saveError && <span className="text-[10px] text-red-600 mr-1">{saveError}</span>}
                          <button type="button" onClick={() => saveEdit(ent.id)} disabled={saving || !editCode.trim()}
                            className="rounded-lg bg-[#1E6F5C] px-3 py-1.5 text-xs font-semibold text-white hover:brightness-110 disabled:opacity-50 flex items-center gap-1">
                            {saving && <Loader2 className="h-3 w-3 animate-spin" />} Save
                          </button>
                          <button type="button" onClick={cancelEdit}
                            className="rounded-lg border border-[var(--border)] px-3 py-1.5 text-xs text-[var(--navy)] hover:bg-[#F3F4F6]">
                            Cancel
                          </button>
                        </div>
                      </td>
                    </>
                  ) : (
                    <>
                      <td className="px-4 py-2.5 text-xs text-[var(--navy)]">
                        <span className="font-medium">{ent.client_name}</span>
                        <span className="ml-1.5 text-muted-foreground">({ent.client_slug})</span>
                      </td>
                      <td className="px-4 py-2.5">
                        <span className={`font-mono text-xs font-bold px-2 py-0.5 rounded ${ent.code_duplicate ? 'bg-amber-100 text-amber-800' : 'bg-[#EFF6FF] text-[#3B82F6]'}`}>
                          {ent.code}
                        </span>
                        {ent.code_duplicate && (
                          <span className="ml-1.5 text-[9px] font-bold text-amber-700 uppercase tracking-wide">Duplicate</span>
                        )}
                      </td>
                      <td className="px-4 py-2.5 text-xs text-[var(--navy)]">
                        {ent.short_name || ent.entity_name}
                      </td>
                      <td className="px-4 py-2.5 text-xs text-muted-foreground">
                        {ent.org_type || (ent.is_consolidated ? 'Consolidated' : '—')}
                      </td>
                      <td className="px-2 py-2.5">
                        <button onClick={() => startEdit(ent)}
                          className="rounded-lg p-1.5 text-muted-foreground hover:bg-[#F3F4F6] hover:text-[var(--navy)]">
                          <Pencil className="h-3.5 w-3.5" />
                        </button>
                      </td>
                    </>
                  )}
                </tr>
              ))}
              {filtered.length === 0 && (
                <tr><td colSpan={5} className="px-4 py-8 text-center text-sm text-muted-foreground">No entities found</td></tr>
              )}
            </tbody>
          </table>
        </div>
      )}
    </div>
  );
}

// ── Query Inbox tab (Deepwoods staff) ────────────────────────────────────────

function SourceThreadsView({
  clientId,
  fy,
  onBack,
}: {
  clientId: number;
  fy: string;
  onBack: () => void;
}) {
  const { data: summary = {}, isLoading } = useNdcThreadsSummary(fy, clientId);
  const [selectedSource, setSelectedSource] = useState<string | null>(null);

  const entries = Object.entries(summary);
  const waitingForDW = entries.filter(([, e]) => !e.has_open && e.count > 0);
  const waiting = entries.filter(([, e]) => e.has_open);

  if (isLoading) {
    return (
      <div className="flex items-center justify-center py-12">
        <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
      </div>
    );
  }

  if (entries.length === 0) {
    return (
      <div className="py-8 text-center text-sm text-muted-foreground">
        No query threads for this client yet.
      </div>
    );
  }

  const Section = ({
    title,
    rows,
  }: {
    title: string;
    rows: [string, { count: number; has_open: boolean }][];
  }) => (
    <div className="space-y-2">
      <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">{title}</p>
      {rows.map(([sourceId, entry]) => (
        <div key={sourceId}>
          <button
            type="button"
            onClick={() => setSelectedSource(s => s === sourceId ? null : sourceId)}
            className="flex w-full items-center gap-3 rounded-lg border bg-white px-3 py-2.5 text-left text-sm shadow-sm hover:bg-neutral-soft/30 transition-colors"
            style={{ borderColor: "var(--border)" }}
          >
            <span className="rounded bg-neutral-soft px-1.5 py-0.5 font-mono text-[11px] text-muted-foreground shrink-0">
              {sourceId}
            </span>
            <span className="flex-1 font-medium text-navy text-xs">
              {entry.count} thread{entry.count !== 1 ? "s" : ""}
            </span>
            {entry.has_open && (
              <span className="rounded-full bg-amber-warn px-2 py-0.5 text-[10px] font-bold text-white">
                Reply needed
              </span>
            )}
            <ChevronRight className="h-3.5 w-3.5 text-muted-foreground" />
          </button>
          {selectedSource === sourceId && (
            <div className="mt-1 ml-3 border-l-2 border-primary/30 pl-3">
              <ThreadPanel
                sourceType={sourceId.startsWith("c") && /^c\d+$/.test(sourceId) ? "scope3" : "scope12"}
                sourceId={sourceId}
                fy={fy}
                clientId={clientId}
              />
            </div>
          )}
        </div>
      ))}
    </div>
  );

  return (
    <div className="space-y-4">
      <button
        type="button"
        onClick={onBack}
        className="flex items-center gap-1.5 text-xs font-medium text-primary hover:underline"
      >
        <ChevronLeft className="h-3.5 w-3.5" /> Back to all clients
      </button>

      {waiting.length > 0 && (
        <Section title="Awaiting client reply" rows={waiting} />
      )}
      {waitingForDW.length > 0 && (
        <Section title="Awaiting Deepwoods reply" rows={waitingForDW} />
      )}
    </div>
  );
}

// ── Data Health Tab ───────────────────────────────────────────────────────────

function DataHealthTab() {
  const { data, isLoading, refetch, dataUpdatedAt } = useDataHealth();

  const severityColor = (s: string) =>
    s === "high" ? "var(--red-500,#ef4444)" : s === "medium" ? "var(--amber-500,#f59e0b)" : "var(--sky-500,#0ea5e9)";

  const statusBg = (status: string, severity: string) => {
    if (status === "ok") return { bg: "#d1fae5", text: "#065f46" };
    if (severity === "high")   return { bg: "#fee2e2", text: "#991b1b" };
    if (severity === "medium") return { bg: "#fef3c7", text: "#92400e" };
    return { bg: "#e0f2fe", text: "#0c4a6e" };
  };

  const openChecks  = data?.checks.filter(c => c.status === "open") ?? [];
  const okChecks    = data?.checks.filter(c => c.status === "ok")   ?? [];
  const updatedAt   = dataUpdatedAt ? new Date(dataUpdatedAt).toLocaleTimeString() : null;

  return (
    <div className="space-y-5">
      {/* Summary strip */}
      <div className="flex flex-wrap items-center gap-3">
        <div className="flex items-center gap-2 rounded-lg border px-4 py-2.5" style={{borderColor:"var(--border)",background:"var(--card)"}}>
          <AlertCircle className="w-4 h-4 text-red-500"/>
          <span className="text-sm font-medium">{data?.summary.open_high ?? "—"}</span>
          <span className="text-xs" style={{color:"var(--navy-2)"}}>high severity open</span>
        </div>
        <div className="flex items-center gap-2 rounded-lg border px-4 py-2.5" style={{borderColor:"var(--border)",background:"var(--card)"}}>
          <AlertCircle className="w-4 h-4 text-amber-500"/>
          <span className="text-sm font-medium">{data?.summary.open_medium ?? "—"}</span>
          <span className="text-xs" style={{color:"var(--navy-2)"}}>medium open</span>
        </div>
        <div className="flex items-center gap-2 rounded-lg border px-4 py-2.5" style={{borderColor:"var(--border)",background:"var(--card)"}}>
          <CheckCircle2 className="w-4 h-4 text-emerald-500"/>
          <span className="text-sm font-medium">{data?.summary.resolved ?? "—"}</span>
          <span className="text-xs" style={{color:"var(--navy-2)"}}>resolved</span>
        </div>
        <button
          onClick={() => refetch()}
          className="ml-auto flex items-center gap-1.5 rounded-lg border px-3 py-2 text-xs font-medium hover:bg-[#F3F4F6]"
          style={{borderColor:"var(--border)"}}
        >
          <RefreshCw className="w-3.5 h-3.5"/>
          Re-run checks{updatedAt ? ` · ${updatedAt}` : ""}
        </button>
      </div>

      {isLoading && (
        <div className="flex items-center gap-2 text-sm py-8 justify-center" style={{color:"var(--navy-2)"}}>
          <Loader2 className="w-4 h-4 animate-spin"/>Running diagnostic queries…
        </div>
      )}

      {/* Open issues */}
      {openChecks.length > 0 && (
        <div className="space-y-2">
          <p className="text-xs font-semibold uppercase tracking-widest" style={{color:"var(--navy-2)"}}>Open Issues ({openChecks.length})</p>
          {openChecks.map(check => {
            const col = statusBg(check.status, check.severity);
            return (
              <div key={check.id} className="rounded-lg border" style={{borderColor:"var(--border)",background:"var(--card)"}}>
                <div className="flex items-start gap-3 px-4 py-3">
                  <div className="mt-0.5 flex-shrink-0">
                    <AlertCircle className="w-4 h-4" style={{color: severityColor(check.severity)}}/>
                  </div>
                  <div className="flex-1 min-w-0">
                    <div className="flex flex-wrap items-center gap-2 mb-1">
                      <span className="text-sm font-medium">{check.check}</span>
                      <span className="rounded-full px-2 py-0.5 text-[11px] font-medium" style={{background:col.bg,color:col.text}}>
                        {check.severity} · {check.count} issue{check.count !== 1 ? "s" : ""}
                      </span>
                      <span className="rounded-full border px-2 py-0.5 text-[11px]" style={{borderColor:"var(--border)",color:"var(--navy-2)"}}>
                        {check.client_name}
                      </span>
                    </div>
                    <p className="text-sm" style={{color:"var(--navy-2)"}}>{check.detail}</p>
                    <p className="mt-1 text-xs font-mono" style={{color:"var(--navy-3)"}}>Fix: {check.fix}</p>
                  </div>
                </div>
              </div>
            );
          })}
        </div>
      )}

      {/* Resolved / OK */}
      {okChecks.length > 0 && (
        <div className="space-y-2">
          <p className="text-xs font-semibold uppercase tracking-widest" style={{color:"var(--navy-2)"}}>Passing Checks ({okChecks.length})</p>
          {okChecks.map(check => (
            <div key={check.id} className="flex items-start gap-3 rounded-lg border px-4 py-3" style={{borderColor:"var(--border)",background:"var(--card)"}}>
              <CheckCircle2 className="w-4 h-4 mt-0.5 flex-shrink-0 text-emerald-500"/>
              <div className="flex-1">
                <div className="flex flex-wrap items-center gap-2">
                  <span className="text-sm font-medium">{check.check}</span>
                  <span className="rounded-full border px-2 py-0.5 text-[11px]" style={{borderColor:"var(--border)",color:"var(--navy-2)"}}>
                    {check.client_name}
                  </span>
                </div>
                <p className="text-sm" style={{color:"var(--navy-2)"}}>{check.detail}</p>
              </div>
            </div>
          ))}
        </div>
      )}

      {!isLoading && !data && (
        <p className="text-sm py-4" style={{color:"var(--navy-2)"}}>Could not load diagnostic data — check backend connectivity.</p>
      )}
    </div>
  );
}

function QueryInboxTab({ clients }: { clients: { id: number; company: string }[] }) {
  const [fy, setFy] = useState("FY26");
  const [selectedClientId, setSelectedClientId] = useState<number | null>(null);

  const { data: allSummary, isLoading: loadingAll } = useNdcAllClientsSummary(fy);

  const clientById = useMemo(() => {
    const m: Record<number, { company: string }> = {};
    for (const c of clients) m[c.id] = { company: c.company };
    return m;
  }, [clients]);

  const clientsWithThreads = useMemo(() => {
    if (!allSummary) return [];
    return Object.entries(allSummary.by_client)
      .map(([id, entry]) => ({ id: Number(id), ...entry, company: clientById[Number(id)]?.company ?? `Client ${id}` }))
      .sort((a, b) => b.open_count - a.open_count);
  }, [allSummary, clientById]);

  const selectedClient = selectedClientId != null
    ? clients.find(c => c.id === selectedClientId) ?? null
    : null;

  return (
    <div className="space-y-4">
      {/* FY selector + total badge */}
      <div className="flex items-center gap-3 flex-wrap">
        <div className="flex items-center gap-2">
          <label className="text-xs font-medium text-muted-foreground">FY</label>
          <select
            value={fy}
            onChange={e => { setFy(e.target.value); setSelectedClientId(null); }}
            className="rounded-md border px-2.5 py-1.5 text-sm"
            style={{ borderColor: "var(--border)" }}
          >
            {FY_OPTIONS.map(y => <option key={y} value={y}>{y}</option>)}
          </select>
        </div>
        {allSummary && (
          <span className="rounded-full bg-amber-warn/10 border border-amber-warn/30 px-3 py-1 text-xs font-semibold text-amber-warn">
            {allSummary.total_open} thread{allSummary.total_open !== 1 ? "s" : ""} awaiting Deepwoods reply
          </span>
        )}
      </div>

      {/* Detail view (when client selected) */}
      {selectedClientId != null && selectedClient && (
        <div className="rounded-xl border bg-white p-4 shadow-sm" style={{ borderColor: "var(--border)" }}>
          <div className="mb-4 flex items-center justify-between">
            <div>
              <p className="text-sm font-bold text-navy">{selectedClient.company}</p>

            </div>
          </div>
          <SourceThreadsView clientId={selectedClientId} fy={fy} onBack={() => setSelectedClientId(null)} />
        </div>
      )}

      {/* Client list (when no client selected) */}
      {selectedClientId == null && (
        <>
          {loadingAll && (
            <div className="flex items-center gap-2 text-sm text-muted-foreground py-4">
              <Loader2 className="h-4 w-4 animate-spin" /> Loading…
            </div>
          )}
          {!loadingAll && clientsWithThreads.length === 0 && (
            <div className="rounded-xl border bg-white py-10 text-center text-sm text-muted-foreground shadow-sm" style={{ borderColor: "var(--border)" }}>
              No query threads found for {fy}.
            </div>
          )}
          <div className="space-y-2">
            {clientsWithThreads.map(c => (
              <button
                key={c.id}
                type="button"
                onClick={() => setSelectedClientId(c.id)}
                className="flex w-full items-center gap-3 rounded-xl border bg-white px-4 py-3 text-left shadow-sm hover:bg-neutral-soft/30 transition-colors"
                style={{ borderColor: "var(--border)" }}
              >
                <div className="flex-1 min-w-0">
                  <p className="text-sm font-semibold text-navy truncate">{c.company}</p>
                  <p className="text-xs text-muted-foreground">{c.count} thread{c.count !== 1 ? "s" : ""} total</p>
                </div>
                {c.open_count > 0 && (
                  <span className="rounded-full bg-amber-warn px-2.5 py-0.5 text-xs font-bold text-white shrink-0">
                    {c.open_count} for DW
                  </span>
                )}
                <ChevronRight className="h-4 w-4 text-muted-foreground shrink-0" />
              </button>
            ))}
          </div>
        </>
      )}
    </div>
  );
}

// ── Drive Settings Tab ────────────────────────────────────────────────────────

interface DriveClientStatus {
  id: number;
  name: string;
  slug: string;
  connected: boolean;
  has_folder: boolean;
  folder_id: string | null;
}

function DriveSettingsTab() {
  const { data: statuses, isLoading, isError, error, refetch } = useQuery<DriveClientStatus[]>({
    queryKey: ["drive-status-all"],
    queryFn: () => authFetch("ghg/drive-status.php?all=1"),
  });

  const [connecting, setConnecting] = useState<number | null>(null);

  async function handleConnect(clientId: number) {
    setConnecting(clientId);
    try {
      const data = await authFetch(`ghg/drive-connect.php?client_id=${clientId}`);
      const { auth_url } = data;
      const popup = window.open(auth_url, "driveConnect", "width=600,height=700,left=300,top=100");
      if (!popup) {
        toast.error("Popup blocked — please allow popups for this site");
        return;
      }
      // Poll until popup closes, then refresh status
      const timer = setInterval(() => {
        if (popup.closed) {
          clearInterval(timer);
          setConnecting(null);
          refetch();
          toast.success("Drive account connected — status refreshed");
        }
      }, 800);
    } catch {
      toast.error("Error initiating Google Drive connection");
    } finally {
      // connecting state cleared when popup closes (or on error)
      if (connecting === clientId) setConnecting(null);
    }
  }

  if (isLoading) {
    return (
      <div className="flex items-center gap-2 py-8 text-sm" style={{ color: "var(--navy-2)" }}>
        <Loader2 className="w-4 h-4 animate-spin" /> Loading Drive status…
      </div>
    );
  }

  if (isError) {
    return (
      <div className="rounded-xl border border-red-200 bg-red-50 px-5 py-4 space-y-2">
        <p className="text-sm font-semibold text-red-700">API error — Drive status could not be loaded</p>
        <pre className="text-xs text-red-600 whitespace-pre-wrap break-all">{String(error)}</pre>
        <button onClick={() => refetch()} className="text-xs underline text-red-600">Retry</button>
      </div>
    );
  }

  return (
    <div className="space-y-4">
      <div>
        <h3 className="font-semibold text-base mb-1" style={{ color: "var(--navy)" }}>Google Drive Connections</h3>
        <p className="text-sm" style={{ color: "var(--navy-2)" }}>
          Connect the appropriate Deepwoods Google account for each client. Each client's uploaded evidence will be stored in their designated Drive folder using the connected account's credentials.
        </p>
      </div>

      <div className="rounded-xl border bg-amber-50 border-amber-200 px-4 py-3 text-sm text-amber-800 flex items-start gap-2">
        <AlertCircle className="w-4 h-4 flex-shrink-0 mt-0.5" />
        <span>
          When connecting, sign in with the <strong>Deepwoods Google account that owns that client's Drive folder</strong>. Different clients may use different Deepwoods accounts.
        </span>
      </div>

      <div className="space-y-2">
        {(statuses ?? []).map(client => (
          <div key={client.id} className="bg-white rounded-xl border shadow-sm px-5 py-4 flex flex-wrap items-center gap-4" style={{ borderColor: "var(--border)" }}>
            <div className="w-9 h-9 rounded-full flex items-center justify-center text-white text-xs font-bold flex-shrink-0" style={{ background: "var(--deepwoods-green)" }}>
              {client.name.slice(0, 2).toUpperCase()}
            </div>
            <div className="flex-1 min-w-0">
              <p className="font-semibold text-sm truncate" style={{ color: "var(--navy)" }}>{client.name}</p>
              <p className="text-xs" style={{ color: "var(--navy-2)" }}>
                {client.has_folder ? `Folder: ${client.folder_id}` : "No root folder configured in DB"}
              </p>
            </div>

            {/* Connection status badge */}
            {client.connected ? (
              <span className="flex items-center gap-1.5 text-[11px] font-semibold px-2.5 py-1 rounded-full bg-[#E6F4EF] text-[#1E6F5C]">
                <Link2 className="w-3 h-3" /> Connected
              </span>
            ) : (
              <span className="flex items-center gap-1.5 text-[11px] font-semibold px-2.5 py-1 rounded-full bg-slate-100 text-slate-500">
                <Link2Off className="w-3 h-3" /> Not connected
              </span>
            )}

            <button
              onClick={() => handleConnect(client.id)}
              disabled={connecting === client.id}
              className="flex items-center gap-1.5 rounded-lg px-3.5 py-2 text-xs font-semibold text-white hover:brightness-110 disabled:opacity-60"
              style={{ background: "var(--deepwoods-green)" }}
            >
              {connecting === client.id ? (
                <><Loader2 className="w-3.5 h-3.5 animate-spin" /> Connecting…</>
              ) : (
                <><HardDriveUpload className="w-3.5 h-3.5" /> {client.connected ? "Reconnect" : "Connect Google Account"}</>
              )}
            </button>
          </div>
        ))}

        {(statuses ?? []).length === 0 && (
          <div className="rounded-xl border bg-white py-10 text-center text-sm shadow-sm" style={{ borderColor: "var(--border)", color: "var(--navy-2)" }}>
            No clients found.
          </div>
        )}
      </div>
    </div>
  );
}

                              {locked && <span className="text-[11px] font-semibold px-2 py-0.5 rounded-full bg-[#E6F4EF] text-[#1E6F5C]">Always relevant</span>}
                              {ansCfg && (
                                <span className="text-[11px] font-semibold px-2 py-0.5 rounded-full border"
                                  style={{background:ansCfg.bg,color:ansCfg.color,borderColor:ansCfg.color+"55"}}>
                                  Client: {ansCfg.label}
                                </span>
                              )}
                              {!ansCfg && clientAnswers && (
                                <span className="text-[11px] px-2 py-0.5 rounded-full bg-slate-100 text-slate-400">Client: —</span>
                              )}
                            </div>
                          </div>
                          <div className="flex gap-1.5 flex-shrink-0">
                            {([["relevant","Relevant","#E6F4EF","#1E6F5C"],["unsure","Not sure","#FEF3C7","#D97706"],["not_relevant","Not relevant","#F3F4F6","#6B7280"]] as const).map(([v,l,bg,col])=>(
                              <button key={v} disabled={locked} onClick={()=>patchRelevance(i,v as Cat3Relevance)}
                                className="px-2.5 py-1 rounded-lg text-[11px] font-semibold border transition-all"
                                style={val===v||(locked&&v==="relevant")?{background:bg,color:col,borderColor:col}:{borderColor:"var(--border)",color:"var(--navy-2)"}}>
                                {l}
                              </button>
                            ))}
                          </div>
                        </div>
                        {/* Notes row */}
                        <div className="px-4 pb-3.5 pt-0">
                          <input
                            type="text"
                            value={note ?? ''}
                            onChange={e => patchNote(i, e.target.value)}
                            placeholder="Add consultant note (optional) — shown to client"
                            className="w-full rounded-lg border px-3 py-1.5 text-xs outline-none focus:border-[#1E6F5C]"
                            style={{borderColor:"var(--border)",color:"var(--navy)",background:"var(--paper)"}}
                          />
                        </div>
                      </div>
                    );
                  })}
                </div>

                <div className="flex gap-3 pt-2">
                  <button onClick={markScreeningDone}
                    className="flex items-center gap-2 rounded-lg px-5 py-2.5 text-sm font-semibold text-white hover:brightness-110" style={{background:"var(--deepwoods-green)"}}>
                    <CheckCircle2 className="w-4 h-4"/> Save & mark screening complete
                  </button>
                  <button onClick={()=>{setScreeningEntity(null);}} className="rounded-lg border px-4 py-2.5 text-sm font-medium hover:bg-[#F3F4F6]" style={{borderColor:"var(--border)",color:"var(--navy)"}}>
                    Back
                  </button>
                  <button onClick={()=>{setSelectedClient(null); setScreeningEntity(null);}} className="rounded-lg border px-4 py-2.5 text-sm font-medium hover:bg-[#F3F4F6]" style={{borderColor:"var(--border)",color:"var(--navy)"}}>
                    Cancel
                  </button>
                </div>
              </>
            )}
          </div>
        )}

        {/* ── Verification tab ── */}
        {tab==="verification" && <VerificationTab />}

        {/* ── Entity Codes tab ── */}
        {tab==="entities" && <EntitiesAdminTab />}

        {/* ── Branding tab ── */}
        {tab==="branding" && <BrandingTab />}

        {/* ── Query Inbox tab ── */}
        {tab==="queries" && (
          <QueryInboxTab
            clients={apiClients.map(c => ({ id: c.id, company: c.name }))}
          />
        )}

        {/* ── Data Health tab ── */}
        {tab==="data-health" && <DataHealthTab />}

        {/* ── Drive Settings tab ── */}
        {tab==="drive" && <DriveSettingsTab />}

        {/* ── Ledger Import tab ── */}
        {tab==="ledger" && <LedgerImportTab clients={apiClients} />}

        {/* ── Documents tab ── */}
        {tab==="documents" && <DocumentsTab clients={apiClients} />}

        {/* ── Data Gaps tab ── */}
        {tab==="data-gaps" && <DataGapsTab clients={apiClients} />}

        {/* ── Audit log tab ── */}
        {tab==="log" && (
          <div className="space-y-3">
            <p className="text-sm" style={{color:"var(--navy-2)"}}>Immutable log of all consultant and system actions.</p>
            {[
              {ts:"2026-07-25 14:32",actor:"admin@deepwoodsgreen.com",action:"Approved submission #1035 — Client B / Cat 1 Purchased Goods"},
              {ts:"2026-07-24 11:08",actor:"System",action:"Report generated for Client B — INTERIM — 13/14 data points"},
              {ts:"2026-07-22 09:45",actor:"admin@deepwoodsgreen.com",action:"Relevance screening completed for Client A"},
              {ts:"2026-07-20 16:12",actor:"System",action:"Submission received — Client A / Scope 3 Cat 1 (procurement Q1–Q3)"},
              {ts:"2026-07-18 10:00",actor:"admin@deepwoodsgreen.com",action:"Client C onboarded — CLIENT-003 created"},
              {ts:"2026-07-15 15:30",actor:"System",action:"Daily cycle trigger ran — 2 reminder emails sent"},
              {ts:"2026-07-10 08:14",actor:"System",action:"Submission received — Client A / Scope 3 Cat 6 Business Travel data"},
              {ts:"2026-06-22 13:00",actor:"admin@deepwoodsgreen.com",action:"Approved submission #1030 — Client E / Scope 1"},
              {ts:"2026-06-15 10:45",actor:"System",action:"Report generated for Client E — FINAL — approved & sent to client"},
            ].map((e,i)=>(
              <div key={i} className="bg-white rounded-xl border px-4 py-3 shadow-sm flex items-start gap-4" style={{borderColor:"var(--border)"}}>
                <ScrollText className="w-4 h-4 flex-shrink-0 mt-0.5" style={{color:"var(--navy-2)"}}/>
                <div className="flex-1 min-w-0">
                  <p className="text-sm" style={{color:"var(--navy)"}}>{e.action}</p>
                  <p className="text-xs mt-0.5" style={{color:"var(--navy-2)"}}>{e.ts} · {e.actor}</p>
                </div>
              </div>
            ))}
          </div>
        )}
      </PageBody>
    </>
  );
}
